Nope, good point. So I
parted /dev/xvdaa mklabel gpt
parted -s /dev/xvdaa mkpart primary 0% 100%
mkfs -t ext4 /dev/xvdaa1
mount -t ext4 /dev/xvdaa1 /mnt
and
root_at_node001:~# df -h
Filesystem Size Used Avail Use% Mounted on
/dev/sda1 7.9G 5.7G 1.8G 77% /
…
master:/home 7.9G 5.8G 1.7G 78% /home
master:/opt/sge6 7.9G 5.8G 1.7G 78% /opt/sge6
/dev/xvdaa1 74G 180M 70G 1% /mnt

USERDATA_SCRIPTS = /root/StarCluster/installR.sh
